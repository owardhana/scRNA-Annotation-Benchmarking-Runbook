# Per-tool conda environments for the Python deep-learning and semi-supervised tools.
#
# These are the install recipes as recorded while the study ran, corrected only where a
# line could not run as written (typos, non-package names, broken version syntax). They
# are the only record of the workstation environments: no environment exports were kept,
# so packages left unpinned here were resolved at install time. The original, uncorrected
# notes are in as-run/benchmarking/python-DL-implementation/Install Notes DL.txt.
#
# Not recorded: scBalance_env, Cell_BLAST_env, MARS_env and the CellTypist environment.
# Their wrappers' imports show what each needs (scBalance, Cell_BLAST, torch; MARS: torch;
# all: anndata, memory_profiler). Recipes for tools with no released wrapper are left out
# here and remain in the original notes: scDeepInsight, scnym, mtANN and ItClust (see
# docs/tool-roster.md), and scNAME, which was set up but is not in the reported panel.

$ conda create -n scDeepSort_env python=3.8
$ conda activate scDeepSort_env
$ pip install /path/to/scDeepSort-v1.0-cu102.tar.gz   # source tarball, not on PyPI
$ conda install conda-forge::anndata openpyxl

    scipy==1.9.3 \
    numpy==1.23.5 \
    scikit-learn==1.1.3 \
    pandas==1.5.3 \
    matplotlib==3.6.3 -y
